## 六、本报告的可复现命令

```bash
# 静态分析（排除第三方）
cppcheck --enable=warning,style,performance,portability --std=c++17 \
  --language=c++ --inline-suppr -q \
  -I core/src -I core/include core/src 2>&1 \
  | grep -v "third_party\|ed25519_verify"

# ≥80 行成员函数枚举
python - <<'EOF'
import io,re
from pathlib import Path
pat=re.compile(r'^([A-Za-z_][\w:<>,\s\*&]*?)\b(\w+)::(\w+)\s*\(')
for p in sorted(Path('core/src').rglob('*.cpp')):
    if 'third_party' in str(p) or 'ed25519' in str(p): continue
    lines=io.open(p,encoding='utf-8').read().splitlines(); i=0
    while i<len(lines):
        m=pat.match(lines[i])
        if m:
            d=0;s=False;j=i
            while j<len(lines):
                for ch in lines[j]:
                    if ch=='{': d+=1;s=True
                    elif ch=='}': d-=1
                if s and d<=0: break
                j+=1
            if j-i+1>=80: print('%5d %s:%d %s()'%(j-i+1,p,i+1,m.group(3)))
            i=j+1
        else: i+=1
EOF

# 死函数核实（生产 0 引用 / 仅测试引用）
grep -rn "\b<函数名>\b" core/src core/tests --include=*.cpp --include=*.hpp
```
